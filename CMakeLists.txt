cmake_minimum_required(VERSION 3.24)
project(ZahnCAD VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ZAHNCAD_DESKTOP "Build Qt/VTK desktop" OFF)
include(CTest)
add_library(zahn_core src/mesh.cpp)
target_include_directories(zahn_core PUBLIC include)
add_executable(zahncad-cli src/cli.cpp)
target_link_libraries(zahncad-cli PRIVATE zahn_core)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE zahn_core)
  add_test(NAME core COMMAND core_tests)
endif()
if(ZAHNCAD_DESKTOP)
  find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Concurrent)
  find_package(LibArchive REQUIRED)
  find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel FiltersCore FiltersGeneral FiltersModeling FiltersExtraction FiltersSources InteractionStyle InteractionWidgets RenderingCore RenderingOpenGL2 RenderingAnnotation IOImage IOPLY GUISupportQt)
  add_library(zahn_surface src/surface.cpp)
  target_link_libraries(zahn_surface PUBLIC zahn_core ${VTK_LIBRARIES})
  add_library(zahn_project src/project.cpp src/document.cpp)
  target_link_libraries(zahn_project PUBLIC zahn_surface Qt6::Widgets LibArchive::LibArchive)
  add_library(zahn_display src/display.cpp src/interaction.cpp)
  target_link_libraries(zahn_display PUBLIC zahn_project ${VTK_LIBRARIES})
  add_executable(zahncad WIN32 src/desktop.cpp)
  target_link_libraries(zahncad PRIVATE zahn_display Qt6::Sql Qt6::Concurrent)
  vtk_module_autoinit(TARGETS zahncad MODULES ${VTK_LIBRARIES})
  if(BUILD_TESTING)
    add_executable(project_tests tests/project_tests.cpp)
    target_link_libraries(project_tests PRIVATE zahn_project)
    add_test(NAME project COMMAND project_tests)
    add_executable(viewer_tests tests/viewer_tests.cpp)
    target_link_libraries(viewer_tests PRIVATE zahn_display Qt6::Concurrent)
    vtk_module_autoinit(TARGETS viewer_tests MODULES ${VTK_LIBRARIES})
    add_test(NAME viewer COMMAND viewer_tests)
    add_executable(surface_tests tests/surface_tests.cpp)
    target_link_libraries(surface_tests PRIVATE zahn_project)
    vtk_module_autoinit(TARGETS surface_tests MODULES ${VTK_LIBRARIES})
    add_test(NAME surface COMMAND surface_tests)
    add_executable(viewer_benchmark tests/viewer_benchmark.cpp)
    target_link_libraries(viewer_benchmark PRIVATE zahn_display)
    vtk_module_autoinit(TARGETS viewer_benchmark MODULES ${VTK_LIBRARIES})
  endif()
  install(TARGETS zahncad RUNTIME DESTINATION bin)
endif()
install(TARGETS zahncad-cli RUNTIME DESTINATION bin)
set(CPACK_PACKAGE_NAME ZahnCAD)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
